            font-size:20px;

<body>

    <h1 id="text">Shrinking and Enlarging Text</h1>

    <script>
        let text = document.getElementById("text");

        let size = 20;
        let change = 2;

        setInterval(function () {

            size = size + change;

            if (size >= 50) {
                change = -2;
                text.innerHTML = "Text Shrinking";
                text.style.color = "red";
            }

            if (size <= 20) {
                change = 2;
                text.innerHTML = "Text Enlarging";
                text.style.color = "green";
            }

            text.style.fontSize = size + "px";

        }, 100);
    </script>
