#include<stdio.h>
struct{
    int u,v,w;
}e[100];
int main() {
    int V,E,d[100],p[100]={0};
    printf("Enter vertices and edges: ");
    scanf("%d%d",&V,&E);
    for(int i=0;i<E;i++)    {
        scanf("%d%d%d",&e[i].u,&e[i].v,&e[i].w);
    }
    for(int i=0;i<V;i++)    {
        d[i]=99999;
    }
    d[0]=0;
    for(int i=1;i<V;i++)  {
        for(int j=0;j<E;j++)   {
            if(d[e[j].v] > d[e[j].u]+e[j].w)   {
                d[e[j].v]=d[e[j].u]+e[j].w;
                p[e[j].v]=e[j].u;
            }
        }
    }
    for(int j=0;j<E;j++) {
        if(d[e[j].v] > d[e[j].u]+e[j].w) {
            printf("Negative cycle detected");
            return 0;
        }
    }
    printf("Distance array: ");
    for(int i=0;i<V;i++) {
        printf("%d ",d[i]);
    }
    printf("\nPredecessor array: ");
    for(int i=0;i<V;i++) {
        printf("%d ",p[i]);
    } return 0;
}
