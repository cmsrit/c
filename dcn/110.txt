#include <stdio.h>

#define MAX 100
#define INF 999999

int main() {
    int g[MAX][MAX];
    int d[MAX], v[MAX] = {0};
    int V, i, j, min, u, s;

    printf("Enter number of vertices: ");
    if (scanf("%d", &V) != 1 || V > MAX) {
        printf("Invalid number of vertices.\n");
        return 1;
    }

    printf("Enter the adjacency matrix (%d x %d):\n", V, V);
    for(i = 0; i < V; i++) {
        for(j = 0; j < V; j++) {
            scanf("%d", &g[i][j]);
        }
    }

    printf("Enter the source vertex (0 to %d): ", V - 1);
    scanf("%d", &s);
    if (s < 0 || s >= V) {
        printf("Invalid source vertex.\n");
        return 1;
    }

    for(i = 0; i < V; i++) {
        d[i] = INF;
    }
    d[s] = 0;

    for(i = 0; i < V - 1; i++) {
        min = INF;
        u = -1;

        for(j = 0; j < V; j++) {
            if(!v[j] && d[j] < min) {
                min = d[j];
                u = j;
            }
        }

        if (u == -1) {
            break;
        }

        v[u] = 1;

        for(j = 0; j < V; j++) {
            if(!v[j] && g[u][j] && d[u] + g[u][j] < d[j]) {
                d[j] = d[u] + g[u][j];
            }
        }
    }

    printf("\nShortest paths from Source Vertex %d:\n", s);
    printf("To Vertex\tDistance\n");
    for(i = 0; i < V; i++) {
        if (d[i] == INF) {
            printf("%d \t\t INF\n", i);
        } else {
            printf("%d \t\t %d\n", i, d[i]);
        }
    }

    return 0;
}
