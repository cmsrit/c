 #include <stdio.h>
int main() {
    int d[8], r[8], c1, c2, c3, e;
    printf("Enter 4 data bits (one by one):\n");
    scanf("%d %d %d %d", &d[3], &d[5], &d[6], &d[7]);
    d[1] = d[3] ^ d[5] ^ d[7];
    d[2] = d[3] ^ d[6] ^ d[7];
    d[4] = d[5] ^ d[6] ^ d[7];
    printf("Encoded Hamming Code: ");
    for(int i = 1; i <= 7; i++){ printf("%d ", d[i]);}
    printf("\n\nEnter 7 received bits:\n");
    for(int i = 1; i <= 7; i++) {scanf("%d", &r[i]);}
    c1 = r[1] ^ r[3] ^ r[5] ^ r[7];
    c2 = r[2] ^ r[3] ^ r[6] ^ r[7];
    c3 = r[4] ^ r[5] ^ r[6] ^ r[7];
    e = (c3 * 4) + (c2 * 2) + c1;
    if(e == 0) {
        printf("\nNo error detected.\n");
    } else {
        printf("\nError detected at position: %d", e);
        r[e] ^= 1; // Flip the corrupted bit directly at its position
        printf("\nCorrected data: ");
        for(int i = 1; i <= 7; i++) printf("%d ", r[i]);
        printf("\n");
    }

    return 0;
}
