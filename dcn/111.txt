i) Character Stuffing

#include <stdio.h>
#include <string.h>

int main() {
    char a[100], sd, ed;

    printf("Enter characters to be stuffed: ");
    scanf("%s", a);

    printf("Enter starting delimiter: ");
    scanf(" %c", &sd);

    printf("Enter ending delimiter: ");
    scanf(" %c", &ed);

    printf("After stuffing: %c", sd);
    for(int i = 0; i < strlen(a); i++) {
        if(a[i] == sd || a[i] == ed) {
            printf("%c", a[i]);
        }
        printf("%c", a[i]);
    }
    printf("%c\n", ed);

    return 0;
}

ii) Bit Stuffing


#include <stdio.h>

int main() {
    int a[100], n, count = 0;

    printf("Enter frame size: ");
    if (scanf("%d", &n) != 1 || n > 100) {
        return 1;
    }

    printf("Enter bits (separated by spaces):\n");
    for(int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    printf("After Bit Stuffing: ");
    for(int i = 0; i < n; i++) {
        printf("%d", a[i]);
        if(a[i] == 1) {
            count++;
        } else {
            count = 0;
        }
        if(count == 5) {
            printf("0");
            count = 0;
        }
    }
    printf("\n");

    return 0;
}
